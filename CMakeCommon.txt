function(copy_if_different _target _source _destination)

	add_custom_command (
		TARGET ${_target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E 
			copy_if_different 
				${_source}
				${_destination}
	)

endfunction()

function(generateVSFilters _sourceFiles _filterRoot)

foreach(srcFile IN ITEMS ${_sourceFiles})
    get_filename_component(srcPath "${srcFile}" PATH)
    string(REPLACE "${_filterRoot}" "" groupPath "${srcPath}")
    string(REPLACE "/" "\\" groupPath "${groupPath}")
    source_group("${groupPath}" FILES "${srcFile}")

endforeach()
endfunction()


function(copy_folder _target _source _destination)

	add_custom_command(TARGET ${_target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E 
			copy_directory 
				${_source}
				${_destination})

endfunction()


function(install_with_vcpkg _libraryName)
	execute_process(
			COMMAND ./vcpkg install ${_libraryName}
			WORKING_DIRECTORY ${vcpkg_SOURCE_DIR}
			RESULT_VARIABLE result
		)
endfunction()

function(install_with_nuget _libraryName _version install_path)

		set(NUGET_EXE "${CMAKE_BINARY_DIR}/_deps/nuget.exe")
		if (NOT EXISTS "${NUGET_EXE}")
			message(STATUS "Downloading nuget.exe")
			file(DOWNLOAD
			https://dist.nuget.org/win-x86-commandline/latest/nuget.exe
			"${NUGET_EXE}"
			SHOW_PROGRESS
		)
		endif()
	
		set(NUGET_PACKAGES_DIR "${CMAKE_BINARY_DIR}/_deps/nuget-packages")
		
		execute_process(
			COMMAND "${NUGET_EXE}"
			install ${_libraryName}
			-Version ${_version}
			-OutputDirectory "${NUGET_PACKAGES_DIR}"
			RESULT_VARIABLE NUGET_RESULT
		)

		if (NOT NUGET_RESULT EQUAL 0)
			message(FATAL_ERROR "NuGet install failed")
		else()
			set(${install_path} "${NUGET_PACKAGES_DIR}/${_libraryName}.${_version}" PARENT_SCOPE)
		endif()


endfunction()
